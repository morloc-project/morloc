type Box a kind=record value :: a
type Wrap a kind=record inner :: Box a
type Tree a kind=record node :: a, kids :: [Tree a]
type Opt a kind=data Nope | Some a
type Todo kind=record title :: Str
type Item kind=data Task Todo | Note Str
type Bar kind=record x :: Int
type Holder kind=record task :: Todo
type Q a b kind=record same :: (a, a), crossed :: (b, a)
