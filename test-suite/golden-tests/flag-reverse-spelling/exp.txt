--- opts help ---
Optional arguments:
  -v, --verbose  verbose on/off
                 type: Bool
                 default: false
  -q, --quiet    Turn --verbose off

--- shortOnly help ---
Optional arguments:
  -v, --verbose  verbose on/off
                 type: Bool
                 default: false
  -q             Turn --verbose off

--- longOnly help ---
Optional arguments:
  -v, --verbose  verbose on/off
                 type: Bool
                 default: false
      --quiet    Turn --verbose off
